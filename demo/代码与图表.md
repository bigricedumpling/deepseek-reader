```ts
type Checkpoint = 'D1a' | 'D1b' | 'D2a' | 'D2b'
interface Score {
  checkpoint: Checkpoint
  value: 0 | 1 | 2 | 3 | 4 | -1
  note: string
}
function average(scores: Score[]): number {
  const valid = scores.filter(s => s.value >= 0)
  return valid.reduce((a, s) => a + s.value, 0) / valid.length
}
```
